meshx_err_t meshx_load_persistent_config ( meshx_config_t const * config)
static

Load persistent read-only configuration from flash and apply it.

This function attempts to load product identification (CID, PID, Name) and device UUID from the meshx_cfg partition. If the partition is missing or the data is invalid, it falls back to the defaults provided in the config struct.

Parameters
configPointer to the initial configuration provided by the application.
Returns
meshx_err_t MESHX_SUCCESS on success, or an error code on fatal failure.
193{
194 meshx_err_t err;
195 uint16_t loaded_cid = 0xFFFF;
196 uint16_t loaded_pid = 0xFFFF;
197 uint8_t loaded_uuid[16] = {0};
198
199 /* Load Persistent Read-Only Configuration */
200 err = meshx_ro_cfg_init(&loaded_cid, &loaded_pid, g_product_name, sizeof(g_product_name), loaded_uuid);
201 if (err != MESHX_SUCCESS) {
202 if (err == MESHX_NOT_FOUND || err == MESHX_ERR_RO_CFG_FORMAT) {
203 MESHX_LOGW(MODULE_ID_COMMON, "No valid read-only config found (err 0x%x). Using legacy defaults.", err);
204
205 // Fallback product name
206 if (g_product_name[0] == '\0') {
207 strncpy(g_product_name, CONFIG_PRODUCT_NAME, sizeof(g_product_name) - 1);
208 g_product_name[sizeof(g_product_name)-1] = '\0';
209 }
210
211 // Fallback CID/PID
212 loaded_cid = CONFIG_CID_ID;
213 loaded_pid = CONFIG_PID_ID;
214 } else {
215 return err;
216 }
217 }
218
219 g_cid = loaded_cid;
220 g_pid = loaded_pid;
221 g_vid = loaded_cid;
222
223 // Check if UUID was loaded (if it's not all zeros)
224 bool uuid_loaded = false;
225 for (int i = 0; i < 16; i++) {
226 if (loaded_uuid[i] != 0) {
227 uuid_loaded = true;
228 break;
229 }
230 }
231
232 if (uuid_loaded) {
233 memcpy(g_uuid, loaded_uuid, 16);
234 MESHX_LOGD(MODULE_ID_COMMON, "Using Persistent UUID from config");
235 } else {
236 /* Fallback to empty UUID (will be generated by stack if needed) */
237 memset(g_uuid, 0, 16);
238 MESHX_LOGI(MODULE_ID_COMMON, "Using Empty UUID (Stack will handle generation)");
239 }
240
241 return MESHX_SUCCESS;
242}
