59 /**
60 * @brief Create the platform-specific model instance.
61 *
62 * For ESP-IDF, this populates the model_id with the combined 32-bit

64 *
65 * @param p_baked_model Pointer to the allocated platform model structure.
66 * @return MESHX_SUCCESS on success.

68 meshx_err_t plat_model_create(MESHX_MODEL* p_baked_model) override {
69 if (!p_baked_model) return MESHX_INVALID_ARG;
70
71 /* Set vendor model ID (4 bytes: CID + ID) */
72 uint32_t vnd_id = MESHX_VND_MODEL_ID_UVP;
73 memcpy((void*)&p_baked_model->model_id, &vnd_id, sizeof(vnd_id));
74
75 /*
76 * Note: Opcode handling and callbacks are registered globally in the port layer
77 * via esp_ble_mesh_register_vendor_model_callback.
78 */
79 this->set_plat_model(p_baked_model);
80 return MESHX_SUCCESS;
81 }
