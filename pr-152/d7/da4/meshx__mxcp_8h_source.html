1/**
2 * @file meshx_mxcp.h
3 * @brief MeshX Command Protocol (MXCP) Definitions
4 *
5 * Unified, single-layer, table-driven command/event protocol for
6 * communication between the MeshX Engine and an external Host MCU.
7 *
8 * Frame Format (identical structure to legacy MXSP):
9 * [SOF][LEN][TYPE][PAYLOAD][CHK][EOF]
10 *
11 * TYPE byte encoding:
12 * Bit 7: Direction (0 = CMD Host→Engine, 1 = EVT Engine→Host)
13 * Bits 6-0: Command/Event ID
14 *
15 * @author Pranjal Chanda
16 * @date 2026-05-19
17 * @copyright Copyright 2024 - 2025 MeshX
18 */
19

306/**
307 * @brief MXCP wire frame for RX-side parsing.
308 *
309 * Layout: [SOF][LEN][TYPE][PAYLOAD...][CHK][EOF]
310 * NOTE: Used for RX-side parsing only. TX uses flat-buffer
311 * serialisation for variable-length wire efficiency.
312 */
