100/**

105 *
106 * @param[in] announce Received beacon announcement payload.
107 * @param[in] src_unicast Unicast address of sender node.
108 * @param[in] rssi Received Signal Strength Indication (dBm).

110 *
111 * @return MESHX_SUCCESS on success, error code otherwise.
112 */
113meshx_err_t meshx_proximity_handle_announce(const meshx_uvp_pair_announce_t *announce,
