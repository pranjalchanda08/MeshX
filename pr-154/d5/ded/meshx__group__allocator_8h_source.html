35typedef enum {

39 MESHX_TOPOLOGY_N_TO_1 = 3, /**< N Switches -> 1 Lamp (Anchor = Lamp) */

43/**
44 * @brief Check if address is in dynamic group range (0xC000 - 0xC3FF).
45 *
46 * @param[in] addr Mesh address to check.
47 * @return true if dynamic group address, false otherwise.

49static inline bool meshx_is_dynamic_group_addr(uint16_t addr) {
50 return (addr >= MESHX_DYNAMIC_GROUP_START && addr <= MESHX_DYNAMIC_GROUP_END);

53/**
54 * @brief Calculate group address from anchor node's unicast address.
55 *
56 * Formula: Group Addr = 0xC000 + (anchor_unicast & 0x03FF)
57 *
58 * @param[in] anchor_unicast Unicast address of anchor node.
59 * @return 16-bit group address in range 0xC000..0xC3FF.
60 */
61uint16_t meshx_group_derive_from_anchor(uint16_t anchor_unicast);

63/**
64 * @brief Determine anchor unicast address based on pairing roles.
65 *
66 * - 1:N (Switch -> Lamps): Producer is the hub => Anchor is Switch.
67 * - N:1 (Switches -> Lamp): Consumer is the hub => Anchor is Lamp.
68 *
69 * @param[in] local_unicast Unicast address of local element.
70 * @param[in] peer_unicast Unicast address of peer element.
71 * @param[in] local_role meshx_element_role_t of local element.
72 * @param[in] peer_role meshx_element_role_t of peer element.
73 * @return Unicast address of resolved anchor node.
74 */
75uint16_t meshx_group_resolve_anchor(uint16_t local_unicast,
76 uint16_t peer_unicast,
77 uint8_t local_role,
78 uint8_t peer_role);

80/**

84 * - N:1 (Switches -> Lamp): Consumer is the hub => Anchor is Consumer (Lamp).
85 *
86 * @param[in] local_unicast Unicast address of local element.
87 * @param[in] peer_unicast Unicast address of peer element.

89 * @param[in] peer_role meshx_element_role_t of peer element.
90 * @param[in] topology meshx_topology_type_t archetype.
91 * @return Unicast address of resolved anchor node.
92 */
93uint16_t meshx_group_resolve_anchor_topology(uint16_t local_unicast,

95 uint8_t local_role,
96 uint8_t peer_role,
