33 class LockGuard
34 {
35 meshx_sem_t *m_sem;
36
37 public:
38 explicit LockGuard(meshx_sem_t *sem) : m_sem(sem)
39 {
40 if (m_sem && m_sem->__sem_handle)
41 {
42 meshx_sem_take(m_sem, 0xFFFFFFFF);
43 }
44 }
45 ~LockGuard()
46 {
47 if (m_sem && m_sem->__sem_handle)
48 {
49 meshx_sem_give(m_sem);
50 }
51 }
52 };

71 /**
72 * @brief Register an element instance.
73 * @param element Pointer to the element interface.
74 */
75 void register_element(meshXElementIF *element)
76 {
77 if (!element)
78 return;
79 LockGuard lock(&registry_sem);
80 element_map[element->get_element_idx()] = element;
81 }

83 /**

89 LockGuard lock(&registry_sem);

93 /**
94 * @brief Find an element instance by its index.
95 * @param element_idx Index of the element.
96 * @return meshXElementIF* Pointer to the element instance, or nullptr if not found.
97 */
98 meshXElementIF *find_element(uint16_t element_idx)
99 {
100 LockGuard lock(&registry_sem);
101 auto it = element_map.find(element_idx);
102 if (it != element_map.end())
103 {
104 return it->second;
105 }
106 return nullptr;
107 }
108
109 /**
110 * @brief Find and cast an element instance.
111 * @tparam T The target element type.
112 * @param element_idx Index of the element.
113 * @param expected_variant The expected variant type discriminator.
114 * @return T* Pointer to the casted element instance, or nullptr if not found or type mismatch.
115 */
116 template <typename T>
117 T *find_and_cast(uint16_t element_idx, meshx_element_type_t expected_variant)
118 {
119 meshXElementIF *base = find_element(element_idx);
120 if (base && base->get_element_variant() == expected_variant)
121 {
122 return static_cast<T *>(base);
123 }
124 return nullptr;
125 }
126
127 /**
128 * @brief Get all registered elements.
129 * @return std::map<uint16_t, meshXElementIF*> A copy of the element map.
130 */
131 std::map<uint16_t, meshXElementIF *> get_all_elements()
132 {
133 LockGuard lock(&registry_sem);
134 return element_map;
135 }
