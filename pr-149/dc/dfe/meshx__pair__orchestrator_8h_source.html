58/**
59 * @brief Initiate an autonomous pairing transaction with a peer element.
60 *
61 * Evaluates existing topology for src_elem_idx. If an existing 1:1 link is present
62 * on func_id, initiates Dynamic Topology Promotion (MESHX_SYS_CMD_TOPOLOGY_UPGRADE)
63 * before dispatching MESHX_SYS_CMD_PAIR_REQUEST to the new peer.
64 *
65 * @param[in] src_elem_idx Local element index initiating the pairing.
66 * @param[in] dst_unicast Target node primary unicast address.
67 * @param[in] dst_elem_idx Target element index on peer node.
68 * @param[in] func_id Functional capability ID to bind.
69 *
70 * @return MESHX_SUCCESS on initiation success, error code otherwise.
71 */
72meshx_err_t meshx_pair_initiate(uint8_t src_elem_idx,
73 uint16_t dst_unicast,
74 uint8_t dst_elem_idx,
75 uint16_t func_id);

77/**
78 * @brief Handle an incoming MESHX_SYS_CMD_PAIR_REQUEST wire frame.
79 *
80 * Validates semantic compatibility, provisions group subscription on local
81 * actuator elements, persists the link record to NVS, and increments link_epoch.
82 *
83 * @param[in] req Received pair request payload.
84 * @param[in] src_unicast Unicast address of sender node.
85 * @param[out] rsp Response payload to transmit back to initiator.
86 *
87 * @return MESHX_SUCCESS on success, error code otherwise.
88 */
89meshx_err_t meshx_pair_handle_request(const meshx_uvp_pair_req_t *req,
90 uint16_t src_unicast,

92
93/**
94 * @brief Handle an incoming MESHX_SYS_RSP_PAIR_RESPONSE wire frame.
95 *
96 * Configures the local model publication address, commits the binding to NVS,
97 * increments link_epoch, and invokes registered completion callbacks.
98 *
99 * @param[in] rsp Received pair response payload.
100 * @param[in] src_unicast Unicast address of respondent node.
101 *
102 * @return MESHX_SUCCESS on success, error code otherwise.
103 */
104meshx_err_t meshx_pair_handle_response(const meshx_uvp_pair_rsp_t *rsp,
105 uint16_t src_unicast);

107/**
108 * @brief Handle an incoming MESHX_SYS_CMD_TOPOLOGY_UPGRADE wire frame.
109 *
110 * Invoked on existing 1:1 peers when a producer promotes to a 1:N group cluster.
111 * Subscribes to the new group address and updates the local link record.
112 *
113 * @param[in] upgrade Received upgrade request payload.
114 * @param[in] src_unicast Unicast address of initiator node.
115 * @param[out] rsp Response acknowledgement to transmit.
116 *
117 * @return MESHX_SUCCESS on success, error code otherwise.
118 */
119meshx_err_t meshx_pair_handle_upgrade(const meshx_uvp_upgrade_req_t *upgrade,
