237/**
238 * @brief Get the active UVP 3-octet Vendor Opcode.
239 *
240 * Calculates the 24-bit Bluetooth Mesh vendor opcode dynamically
241 * from the active Company ID: ((0xC00000 | (0x01 << 16)) | CID).
242 *
243 * @return 24-bit calculated Vendor Opcode.
244 */
245static inline uint32_t meshx_uvp_get_opcode(void)
246{
247 return MESHX_UVP_CALC_OPCODE(meshx_get_cid());
248}
249
250/**
251 * @brief Get the active 32-bit Vendor Model Identifier.
252 *
253 * Combines the UVP Model ID (0x0001) and the active Company ID:
254 * ((0x0001 << 16) | CID).
255 *
256 * @return 32-bit composite Vendor Model ID.
257 */
258static inline uint32_t meshx_uvp_get_vnd_model_id(void)
259{
260 return MESHX_CALC_VND_MODEL_ID(meshx_get_cid(), MESHX_MODEL_ID_UVP);
261}
