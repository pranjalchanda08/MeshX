217 /* 1. CID (Bytes 0..1, Big-Endian) */

226 uuid[9] = (uint8_t)(pid & 0xFF);

229 memcpy(uuid + 10, CONFIG_UUID_FOOTER, 6);

234 uuid[0], uuid[1], uuid[2], uuid[3], uuid[4], uuid[5], uuid[6],
235 uuid[7], uuid[8], uuid[9], uuid[10], uuid[11], uuid[12], uuid[13],
236 uuid[14], uuid[15]);
