241/**
242 * @brief Canonical 8-byte Element Data Message Header.
243 *
244 * Single source of truth for both internal element messages and MXCP wire payloads
245 * (MXCP_CMD_EL_SEND 0x10, MXCP_EVT_EL_DATA_RX_NOTIFY 0x10, MXCP_EVT_EL_DATA_TX_NOTIFY 0x11).
246 */
247typedef struct meshx_data_msg_hdr
248{
249 uint16_t element_id; /**< Element index (0 .. N-1) */
250 uint16_t element_type; /**< Element type (meshx_element_type_t) */
251 uint16_t func_id; /**< Functionality / Opcode ID */
252 uint16_t msg_len; /**< Length of payload following header in bytes */
253} meshx_data_msg_hdr_t;
254
255/**
256 * @brief Canonical Control Message Header.
257 */
258typedef struct meshx_ctrl_msg_hdr
259{
260 uint16_t evt_id; /**< Control Event ID (meshx_ctrl_evt_t) */
261 uint16_t status; /**< Status / Result code (0 = SUCCESS) */
262 uint16_t msg_len; /**< Length of control payload in bytes */
263} meshx_ctrl_msg_hdr_t;
